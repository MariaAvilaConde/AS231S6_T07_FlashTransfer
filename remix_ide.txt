pragma solidity >=0.7.0 <0.9.0;

/// @title Simulador de Transacción
/// @author Juan Condori :D
/// @notice Este contrato permite simular transacciones entre direcciones y gestionar fondos

contract Transaction {
    
    /// @notice Constructor payable que permite enviar ETH al desplegar el contrato
    constructor() payable {}
    
    /// @notice Función para depositar ETH en el contrato
    /// @dev Esta función permite que cualquiera deposite ETH en el contrato
    function deposit() public payable {
        require(msg.value > 0, "Debe enviar algo de ETH");
    }
    
    /// @notice Obtiene el balance actual del contrato
    /// @return El balance del contrato en Wei
    function getContractBalance() public view returns (uint256) {
        return address(this).balance;
    }
    
    /// @notice Envía una transacción a la dirección especificada
    /// @param _recipient La dirección del destinatario
    /// @param _amount La cantidad de Wei a enviar
    function sendTransaction(address payable _recipient, uint256 _amount) public {
        // Verifica que la cantidad a enviar sea mayor que cero
        require(_amount > 0, "La cantidad debe ser mayor que cero");
        
        // Verifica que el contrato tenga suficiente saldo
        require(address(this).balance >= _amount, "Saldo insuficiente en el contrato");

        // Envía la cantidad especificada al destinatario
        _recipient.transfer(_amount);
    }

    /// @notice Permite al contrato recibir Ether directamente
    receive() external payable {}
}
